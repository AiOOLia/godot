# 卫星示例：创建轨道立方体与光源
var angle := 0.0
var satellite: MeshInstance3D
var viewer: Node

func run(v: Node) -> void:
    viewer = v
    var mat := StandardMaterial3D.new()
    mat.albedo_color = Color(0.17, 0.72, 0.6)
    mat.emission_enabled = true
    mat.emission = Color(0.1, 0.4, 0.35)
    satellite = MeshInstance3D.new()
    satellite.mesh = BoxMesh.new()
    satellite.material_override = mat
    add_child(satellite)
    print("卫星已创建")

func _process(delta: float) -> void:
    angle += delta * 0.6
    satellite.position = Vector3(
        cos(angle) * 2.4, 0.8, sin(angle) * 2.4)
    satellite.rotate_y(delta)
